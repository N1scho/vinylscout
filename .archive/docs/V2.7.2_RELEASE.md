```javascript
const renderCameraView = () => (
  <div style={{ /* 10 properties */ }}>

    <div style={{ /* button container */ }}>
      <button onClick={() => {
        if (!anthropicToken) {
          showToast('Please enter API key...', 'error');
          return;
        }
        if (!isCameraActive) {
          showToast('Camera is not active...', 'error');
          return;

        captureAndAnalyze();
      }} style={{ /* 12 properties */ }}>
        <div style={{ /* 4 properties */ }} />
      </button>
    </div>

    {cameraError && (
      <div style={{ /* 7 properties */ }}>
        {cameraError}
      </div>
    )}
  </div>

```

**After** (24 lines):
```javascript
const renderCameraView = () => {
  const handleCapture = () => {
    if (!anthropicToken) {
      showToast('Please enter your Anthropic API key in Settings to use camera identification', 'error');
      return;
    }
    if (!isCameraActive) {
      showToast('Camera is not active. Please allow camera access.', 'error');
      return;
    }
    captureAndAnalyze();
  };

  return (
    <CameraView
      videoRef={videoRef}
      canvasRef={canvasRef}
      isAnalyzing={isAnalyzing}
      cameraError={cameraError}
      onCapture={handleCapture}
      themes={themes}
    />
  );
};
```
