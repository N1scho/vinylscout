```javascript
// hooks/useDiscogsSearch.js
import { useQuery } from '@tanstack/react-query';
import { useDiscogs } from './useDiscogs';

export function useDiscogsSearch(query, page = 1, enabled = true) {
  const { search } = useDiscogs();

  return useQuery({
    queryKey: ['discogs', 'search', query, page],
    queryFn: () => search(query, page),
    enabled: enabled && query.length > 0,
    staleTime: 5 * 60 * 1000, // Cache 5 minutes
    cacheTime: 30 * 60 * 1000, // Keep in cache 30 minutes
    retry: 2,
    refetchOnWindowFocus: false
  });
}

// Usage in SearchView
function SearchView() {
  const [query, setQuery] = useState('');
  const { data, isLoading, error } = useDiscogsSearch(query);

  return (
    <div>
      <input value={query} onChange={(e) => setQuery(e.target.value)} />
      {isLoading && <Spinner />}
      {error && <Error message={error.message} />}
      {data && <Results items={data.results} />}
    </div>
  );
}
```

```javascript
import { useVirtualizer } from '@tanstack/react-virtual';

function CollectionGrid({ items }) {
  const parentRef = useRef();

  const virtualizer = useVirtualizer({
    count: items.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 320, // Card height
    overscan: 5, // Render 5 extra items
    lanes: 3 // 3 columns
  });

  return (
    <div ref={parentRef} style={{ height: '100vh', overflow: 'auto' }}>
      <div
        style={{
          height: `${virtualizer.getTotalSize()}px`,
          position: 'relative'
        }}
      >
        {virtualizer.getVirtualItems().map((virtualRow) => {
          const vinyl = items[virtualRow.index];
          return (
            <div
              key={virtualRow.key}
              style={{
                position: 'absolute',
                top: 0,
                left: 0,
                transform: `translateY(${virtualRow.start}px)`
              }}
            >
              <VinylCard vinyl={vinyl} />
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

#### API Routes

```javascript
// api/collection/[userId].js
import { db } from '../../db';
import { vinyls } from '../../db/schema';
import { eq } from 'drizzle-orm';

export default async function handler(req, res) {
  const { userId } = req.query;

  if (req.method === 'GET') {
    // Get user's collection
    const collection = await db
      .select()
      .from(vinyls)
      .where(eq(vinyls.userId, userId))
      .orderBy(vinyls.addedAt);

    return res.json(collection);
  }

  if (req.method === 'POST') {
    // Add vinyl to collection
    const vinyl = req.body;
    const inserted = await db.insert(vinyls).values({
      ...vinyl,
      userId
    }).returning();

    return res.json(inserted[0]);
  }
}
```

```javascript
      return res.json();
    },

    }
  });

}
```

```javascript
async function syncCollection() {
  const pendingChanges = await getPendingChanges();

      await fetch('/api/collection', {
        method: 'POST',
        body: JSON.stringify(change)
      });
```

```javascript
// hooks/useInstallPrompt.js
export function useInstallPrompt() {
  const [installPrompt, setInstallPrompt] = useState(null);
  const [isInstalled, setIsInstalled] = useState(false);

  useEffect(() => {
    const handler = (e) => {
      e.preventDefault();
      setInstallPrompt(e);
    };

    window.addEventListener('beforeinstallprompt', handler);
    return () => window.removeEventListener('beforeinstallprompt', handler);
  }, []);

  const install = async () => {
    if (!installPrompt) return;

    installPrompt.prompt();
    const { outcome } = await installPrompt.userChoice;

    if (outcome === 'accepted') {
      setIsInstalled(true);
    }

    setInstallPrompt(null);
  };

  return { canInstall: !!installPrompt, install, isInstalled };
}
```

## 💡 Key Architectural Principles

### 1. Component Composition

```javascript
// ❌ Bad: Monolithic
function CollectionView() {
  return (
    <div>
      {/* 500 lines of JSX */}
    </div>
  );
}

// ✅ Good: Composed
function CollectionView() {
  return (
    <div>
      <CollectionHeader />
      <CollectionFilters />
      <CollectionGrid />
      <CollectionPagination />
    </div>
  );
}
```

### 2. Single Responsibility

```javascript
// ❌ Bad: Does everything
function VinylCard({ vinyl, onAdd, onRemove, onFavorite, onShare, onEdit }) {
  // 200 lines of code
}

// ✅ Good: One thing well
function VinylCard({ vinyl, onAction }) {
  return (
    <Card>
      <CardImage src={vinyl.cover} />
      <CardTitle>{vinyl.title}</CardTitle>
      <CardActions onAction={onAction} />
    </Card>
  );
}
```

### 3. Props Over Context (When Possible)

```javascript
// ❌ Avoid: Global context for everything
const ThemeContext = createContext();
const UserContext = createContext();
const CollectionContext = createContext();
const SettingsContext = createContext();
// Context hell

// ✅ Better: Props for most things, context for truly global
function App() {
  const themes = useTheme(); // Context for theme
  return <SearchView themes={themes} />; // Props for data
}
```

### 4. Custom Hooks for Logic

```javascript
// ❌ Bad: Logic in component
function CollectionView() {
  const [collection, setCollection] = useState([]);
  const [isLoading, setIsLoading] = useState(false);

  useEffect(() => {
    setIsLoading(true);
    fetch('/api/collection')
      .then(res => res.json())
      .then(data => setCollection(data))
      .finally(() => setIsLoading(false));
  }, []);

  // 100 more lines...
}

// ✅ Good: Hook extracts logic
function useCollection() {
  // All logic here
  return { collection, isLoading, addVinyl, removeVinyl };
}

function CollectionView() {
  const { collection, isLoading } = useCollection();
  return <div>{/* just UI */}</div>;
}
```
